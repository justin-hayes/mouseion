package epub

import (
	"archive/zip"
	"bytes"
	"testing"
)

func TestPoetryAuditExtraction(t *testing.T) {
	cases := []struct{ name, body, want string }{
		{"br_and_empty_break", "<p>Alpha<br/>Beta<br/><br/>Gamma</p>", "Alpha\n\nBeta\n\nGamma"},
		{"paragraphs_same_output", "<p>Alpha</p><p>Beta</p><p>Gamma</p>", "Alpha\n\nBeta\n\nGamma"},
		{"pre_literal_lines", "<pre>Alpha\nBeta\n\nGamma</pre>", "Alpha Beta Gamma"},
		{"attribute_only_lines", `<p><span class="line">Alpha</span> <span class="line">Beta</span></p>`, "Alpha Beta"},
		{"adjacent_spans", `<p><span class="line">Alpha</span><span class="line">Beta</span></p>`, "AlphaBeta"},
		{"speaker_and_direction", `<div class="speech"><p class="speaker">Faust</p><p class="stage">(tritt auf)</p><p>Alpha</p></div>`, "Faust\n\n(tritt auf)\n\nAlpha"},
	}
	for _, c := range cases {
		t.Run(c.name, func(t *testing.T) {
			var buf bytes.Buffer
			w := zip.NewWriter(&buf)
			f, err := w.Create("probe.xhtml")
			if err != nil {
				t.Fatal(err)
			}
			if _, err = f.Write([]byte("<html><body>" + c.body + "</body></html>")); err != nil {
				t.Fatal(err)
			}
			if err = w.Close(); err != nil {
				t.Fatal(err)
			}
			r, err := zip.NewReader(bytes.NewReader(buf.Bytes()), int64(buf.Len()))
			if err != nil {
				t.Fatal(err)
			}
			got, _, err := extractXHTML(r.File[0])
			if err != nil {
				t.Fatal(err)
			}
			if got != c.want {
				t.Fatalf("got %q want %q", got, c.want)
			}
			t.Logf("body=%q extracted=%q", c.body, got)
		})
	}
}
