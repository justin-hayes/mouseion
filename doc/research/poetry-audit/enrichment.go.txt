package enrichment

import (
	"encoding/json"
	"testing"
)

func TestPoetryAuditProviderPayload(t *testing.T) {
	text := "Der Baum\nsteht heute still im Garten."
	input := TranslationRequest{Language: "de", TargetLanguage: "en", CanonicalLemma: "baum", UPOS: "NOUN", TargetWord: "Baum", ExampleSentence: text, RequireContextualGloss: true}
	codec, err := NewTranslationCodec(LLMConfig{Model: "audit-no-network"})
	if err != nil {
		t.Fatal(err)
	}
	body, err := codec.EncodeRequest(input)
	if err != nil {
		t.Fatal(err)
	}
	var chat struct {
		Messages []struct {
			Content string `json:"content"`
		} `json:"messages"`
	}
	if err = json.Unmarshal(body, &chat); err != nil {
		t.Fatal(err)
	}
	var payload map[string]any
	if err = json.Unmarshal([]byte(chat.Messages[1].Content), &payload); err != nil {
		t.Fatal(err)
	}
	if payload["example_sentence"] != text {
		t.Fatalf("payload=%#v", payload)
	}
	allowed := map[string]bool{}
	for _, field := range ContextualGlossPayloadFields() {
		allowed[field] = true
	}
	for field := range payload {
		if !allowed[field] {
			t.Fatalf("unexpected field %q", field)
		}
	}
	if len(payload) != 7 {
		t.Fatalf("unexpected fields (empty candidate_senses is omitted): %#v", payload)
	}
	t.Logf("contextual payload fields=%v; exact example_sentence=%q", ContextualGlossPayloadFields(), payload["example_sentence"])
	if SentenceHash(text) == SentenceHash("Der Baum steht heute still im Garten.") {
		t.Fatal("newline cache collision")
	}
	t.Log("SentenceHash distinguishes interior newline from space; no external request made")
}
