package cardexport

// Scratch probe, injected through a go -overlay; not part of the repository.
// Runs the real scoreSentenceQuality (text criteria plus German GDEX, since
// tokens are supplied) over a corpus exported from the production Producer.

import (
	"encoding/json"
	"os"
	"testing"

	"github.com/justin-hayes/mouseion/internal/analyzer"
)

type probeToken struct {
	Surface        string            `json:"surface"`
	RawLemma       string            `json:"raw_lemma"`
	CanonicalLemma string            `json:"canonical_lemma"`
	UPOS           string            `json:"upos"`
	Dependency     string            `json:"dependency"`
	Head           uint32            `json:"head"`
	Morphology     map[string]string `json:"morphology"`
	Start          uint64            `json:"start"`
	End            uint64            `json:"end"`
}

type probeSentence struct {
	Text   string       `json:"text"`
	Start  uint64       `json:"start"`
	Tokens []probeToken `json:"tokens"`
}

type probeCase struct {
	Label     string          `json:"label"`
	Language  string          `json:"language"`
	Sentences []probeSentence `json:"sentences"`
}

type probeResult struct {
	Case     string   `json:"case"`
	Sentence string   `json:"sentence"`
	Target   string   `json:"target"`
	Accepted bool     `json:"accepted"`
	Score    int      `json:"score"`
	GDEX     float64  `json:"gdex"`
	Reasons  []string `json:"reasons"`
}

func TestProbeGate(t *testing.T) {
	input, output := os.Getenv("PROBE_CORPUS"), os.Getenv("PROBE_OUT")
	if input == "" || output == "" {
		t.Skip("PROBE_CORPUS and PROBE_OUT are not set")
	}
	raw, err := os.ReadFile(input)
	if err != nil {
		t.Fatal(err)
	}
	var cases []probeCase
	if err := json.Unmarshal(raw, &cases); err != nil {
		t.Fatal(err)
	}
	results := []probeResult{}
	for _, c := range cases {
		for _, s := range c.Sentences {
			tokens := make([]analyzer.Token, 0, len(s.Tokens))
			seen := map[string]bool{}
			targets := []string{}
			for _, pt := range s.Tokens {
				tokens = append(tokens, analyzer.Token{
					Surface:        pt.Surface,
					RawLemma:       pt.RawLemma,
					CanonicalLemma: pt.CanonicalLemma,
					UPOS:           pt.UPOS,
					Dependency:     pt.Dependency,
					Head:           pt.Head,
					Morphology:     pt.Morphology,
					Location:       analyzer.SourceLocation{StartOffset: pt.Start, EndOffset: pt.End},
				})
				switch pt.UPOS {
				case "NOUN", "VERB", "ADJ", "ADV":
					if !seen[pt.Surface] {
						seen[pt.Surface] = true
						targets = append(targets, pt.Surface)
					}
				}
			}
			for _, target := range targets {
				q := scoreSentenceQuality(c.Language, s.Text, target, int64(s.Start), tokens, s.Start)
				results = append(results, probeResult{
					Case: c.Label, Sentence: s.Text, Target: target, Accepted: q.Accepted,
					Score: q.Score, GDEX: q.GDEXScore, Reasons: q.Reasons,
				})
			}
		}
	}
	encoded, err := json.MarshalIndent(results, "", " ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(output, encoded, 0o600); err != nil {
		t.Fatal(err)
	}
	t.Logf("wrote %d (sentence,target) gate decisions", len(results))
}
